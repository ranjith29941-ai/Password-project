
package com.mycompany.mavenproject1;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

import java.util.Arrays;
import java.util.List;

public class PasswordStrengthChecker extends Application {

    private final PasswordField passwordField = new PasswordField();
    private final TextField visibleField = new TextField();
    private final CheckBox showCheck = new CheckBox("Show password");
    private final Button checkButton = new Button("Check Strength");
    private final ProgressBar bar = new ProgressBar(0);

    private final Label strengthLabel = new Label("Strength: -");
    private final Label lengthRule = new Label();
    private final Label upperRule = new Label();
    private final Label lowerRule = new Label();
    private final Label numberRule = new Label();
    private final Label specialRule = new Label();
    private final Label suggestionLabel = new Label();

    private final List<String> commonPasswords = Arrays.asList(
        "password", "123456", "12345678", "qwerty",
        "abc123", "password123", "admin",
        "welcome", "iloveyou", "111111"
    );

    @Override
    public void start(Stage stage) {

        Label title = new Label("Password Strength Checker");
        title.setStyle(
            "-fx-font-size: 22px; -fx-font-weight: bold;"
        );

        passwordField.setPromptText("Enter your password");
        visibleField.setPromptText("Enter your password");

        visibleField.textProperty().bindBidirectional(
            passwordField.textProperty()
        );

        visibleField.visibleProperty().bind(
            showCheck.selectedProperty()
        );
        visibleField.managedProperty().bind(
            showCheck.selectedProperty()
        );

        passwordField.visibleProperty().bind(
            showCheck.selectedProperty().not()
        );
        passwordField.managedProperty().bind(
            showCheck.selectedProperty().not()
        );

        bar.setPrefWidth(350);
        strengthLabel.setStyle("-fx-font-size: 16px; -fx-font-weight: bold;");
        suggestionLabel.setWrapText(true);

        checkButton.setOnAction(e ->
            checkPassword(passwordField.getText())
        );

        passwordField.setOnAction(e ->
            checkPassword(passwordField.getText())
        );

        visibleField.setOnAction(e ->
            checkPassword(passwordField.getText())
        );

        passwordField.textProperty().addListener(
            (obs, oldText, newText) -> resetResult()
        );

        VBox root = new VBox(
            10,
            title,
            passwordField,
            visibleField,
            showCheck,
            checkButton,
            bar,
            strengthLabel,
            lengthRule,
            upperRule,
            lowerRule,
            numberRule,
            specialRule,
            suggestionLabel
        );

        root.setPadding(new Insets(25));
        root.setAlignment(Pos.TOP_LEFT);

        Scene scene = new Scene(root, 420, 520);

        stage.setTitle("Password Strength Checker");
        stage.setScene(scene);
        stage.setResizable(false);
        stage.show();

        resetResult();
    }

    private void checkPassword(String pwd) {

        boolean hasLength = pwd.length() >= 8;
        boolean longLength = pwd.length() >= 12;
        boolean hasUpper = false;
        boolean hasLower = false;
        boolean hasDigit = false;
        boolean hasSpecial = false;

        for (char c : pwd.toCharArray()) {
            if (Character.isUpperCase(c)) {
                hasUpper = true;
            } else if (Character.isLowerCase(c)) {
                hasLower = true;
            } else if (Character.isDigit(c)) {
                hasDigit = true;
            } else if (!Character.isWhitespace(c)) {
                hasSpecial = true;
            }
        }

        if (pwd.isEmpty()) {
            resetResult();
            suggestionLabel.setText(
                "Please enter a password first."
            );
            return;
        }

        setRule(lengthRule, hasLength, "At least 8 characters");
        setRule(upperRule, hasUpper, "Uppercase letter (A-Z)");
        setRule(lowerRule, hasLower, "Lowercase letter (a-z)");
        setRule(numberRule, hasDigit, "Number (0-9)");
        setRule(specialRule, hasSpecial, "Special character (@, #, $)");

        int score = 0;

        if (hasLength) score++;
        if (longLength) score++;
        if (hasUpper) score++;
        if (hasLower) score++;
        if (hasDigit) score++;
        if (hasSpecial) score++;

        boolean isCommon = commonPasswords.contains(
            pwd.toLowerCase(java.util.Locale.ROOT)
        );

        if (isCommon) {
            score = 1;
        }

        String level;
        String color;

        if (score <= 3) {
            level = "Weak";
            color = "red";
        } else if (score <= 5) {
            level = "Medium";
            color = "orange";
        } else {
            level = "Strong";
            color = "green";
        }

        bar.setProgress(score / 6.0);
        bar.setStyle("-fx-accent: " + color + ";");

        strengthLabel.setText("Strength: " + level);
        strengthLabel.setStyle(
            "-fx-font-size: 16px; -fx-font-weight: bold; "
            + "-fx-text-fill: " + color + ";"
        );

        StringBuilder tips = new StringBuilder();

        if (isCommon) {
            tips.append("• Avoid common passwords.\n");
        }
        if (!hasLength) {
            tips.append("• Use at least 8 characters.\n");
        }
        if (!hasUpper) {
            tips.append("• Add an uppercase letter (A-Z).\n");
        }
        if (!hasLower) {
            tips.append("• Add a lowercase letter (a-z).\n");
        }
        if (!hasDigit) {
            tips.append("• Add a number (0-9).\n");
        }
        if (!hasSpecial) {
            tips.append("• Add a special character (@, #, $).\n");
        }
        if (hasLength && !longLength) {
            tips.append("• Use 12 or more characters for better strength.\n");
        }

        if (tips.length() == 0) {
            suggestionLabel.setText(
                "Great! This password meets all the checks."
            );
        } else {
            suggestionLabel.setText(
                "Suggestions:\n" + tips.toString().trim()
            );
        }
    }

    private void resetResult() {
        bar.setProgress(0);
        bar.setStyle("");

        strengthLabel.setText("Strength: -");
        strengthLabel.setStyle(
            "-fx-font-size: 16px; -fx-font-weight: bold;"
        );

        lengthRule.setText("");
        upperRule.setText("");
        lowerRule.setText("");
        numberRule.setText("");
        specialRule.setText("");

        suggestionLabel.setText(
            "Type a password and click Check Strength."
        );
    }

    private void setRule(
        Label label, boolean passed, String text
    ) {
        if (passed) {
            label.setText("\u2713  " + text);
            label.setStyle("-fx-text-fill: green;");
        } else {
            label.setText("\u2717  " + text);
            label.setStyle("-fx-text-fill: red;");
        }
    }

    public static void main(String[] args) {
        launch(args);
    }
}
